create table public.mwd_profiles (
  user_id uuid primary key references auth.users(id) on delete cascade,
  display_name text not null default 'Usuario MWD'
    check (char_length(display_name) between 1 and 80),
  created_at timestamptz not null default now()
);

create table public.mwd_wallets (
  user_id uuid primary key references auth.users(id) on delete cascade,
  account_code text not null unique
    default ('MWD-' || upper(substr(replace(gen_random_uuid()::text, '-', ''), 1, 16))),
  balance_cents bigint not null default 1000000
    check (balance_cents between 0 and 100000000000),
  created_at timestamptz not null default now()
);

create table public.mwd_operations (
  id uuid primary key default gen_random_uuid(),
  actor_id uuid not null references auth.users(id) on delete cascade,
  request_id uuid not null,
  kind text not null check (kind in ('transfer', 'contribution')),
  source_user_id uuid not null references auth.users(id) on delete cascade,
  destination_user_id uuid references auth.users(id) on delete cascade,
  amount_cents bigint not null check (amount_cents between 1 and 100000000),
  created_at timestamptz not null default now(),
  constraint mwd_operation_shape check (
    (kind = 'transfer' and destination_user_id is not null
      and source_user_id <> destination_user_id)
    or (kind = 'contribution' and destination_user_id is null)
  ),
  constraint mwd_unique_request unique (actor_id, request_id)
);
create index mwd_operations_source_time_idx on public.mwd_operations(source_user_id, created_at desc);
create index mwd_operations_destination_time_idx on public.mwd_operations(destination_user_id, created_at desc);

-- Authenticated user moves simulated credits ONLY, with database transaction locks.
create function public.mwd_transfer_demo(
  p_recipient_code text,
  p_amount_cents bigint,
  p_request_id uuid
)
returns jsonb
language plpgsql security definer set search_path = ''
as $$
declare
  v_actor uuid := auth.uid();
  v_destination uuid;
  v_locked integer;
  v_balance bigint;
  v_destination_balance bigint;
  v_existing public.mwd_operations%rowtype;
  v_operation_id uuid;
begin
  if v_actor is null then
    raise exception 'Debes iniciar sesion';
  end if;
  if p_request_id is null or p_amount_cents is null
    or p_amount_cents < 1 or p_amount_cents > 100000000 then
    raise exception 'Solicitud o cantidad invalida';
  end if;
  if p_recipient_code is null or char_length(btrim(p_recipient_code)) > 40 then
    raise exception 'Codigo de destino invalido';
  end if;

  select w.user_id into v_destination
  from public.mwd_wallets w
  where w.account_code = upper(btrim(p_recipient_code));
  if v_destination is null or v_destination = v_actor then
    raise exception 'Cuenta destinataria no disponible';
  end if;

  perform w.user_id from public.mwd_wallets w
  where w.user_id in (v_actor, v_destination)
  order by w.user_id for update;
  get diagnostics v_locked = row_count;
  if v_locked <> 2 then
    raise exception 'Cuenta de prueba no disponible';
  end if;

  select * into v_existing from public.mwd_operations o
  where o.actor_id = v_actor and o.request_id = p_request_id;
  if found then
    if v_existing.kind <> 'transfer' or v_existing.source_user_id <> v_actor
      or v_existing.destination_user_id <> v_destination
      or v_existing.amount_cents <> p_amount_cents then
      raise exception 'Identificador ya usado para otra operacion';
    end if;
    select w.balance_cents into v_balance from public.mwd_wallets w where w.user_id = v_actor;
    return jsonb_build_object('status', 'already_processed',
      'operation_id', v_existing.id, 'balance_cents', v_balance);
  end if;

  select w.balance_cents into v_balance from public.mwd_wallets w where w.user_id = v_actor;
  select w.balance_cents into v_destination_balance
    from public.mwd_wallets w where w.user_id = v_destination;
  if v_balance < p_amount_cents then
    raise exception 'Saldo ficticio insuficiente';
  end if;
  if v_destination_balance > 100000000000 - p_amount_cents then
    raise exception 'Limite de saldo ficticio alcanzado';
  end if;

  update public.mwd_wallets
    set balance_cents = balance_cents - p_amount_cents
    where user_id = v_actor;
  update public.mwd_wallets
    set balance_cents = balance_cents + p_amount_cents
    where user_id = v_destination;

  insert into public.mwd_operations
    (actor_id, request_id, kind, source_user_id, destination_user_id, amount_cents)
    values (v_actor, p_request_id, 'transfer', v_actor, v_destination, p_amount_cents)
    returning id into v_operation_id;

  return jsonb_build_object('status', 'ok',
    'operation_id', v_operation_id, 'balance_cents', v_balance - p_amount_cents);
end
$$;

create function public.mwd_contribute_demo(
  p_amount_cents bigint,
  p_request_id uuid
)
returns jsonb
language plpgsql security definer set search_path = ''
as $$
declare
  v_actor uuid := auth.uid();
  v_balance bigint;
  v_existing public.mwd_operations%rowtype;
  v_operation_id uuid;
begin
  if v_actor is null then
    raise exception 'Debes iniciar sesion';
  end if;
  if p_request_id is null or p_amount_cents is null
    or p_amount_cents < 1 or p_amount_cents > 100000000 then
    raise exception 'Solicitud o cantidad invalida';
  end if;

  select w.balance_cents into v_balance from public.mwd_wallets w
    where w.user_id = v_actor for update;
  if not found then
    raise exception 'Cuenta de prueba no disponible';
  end if;

  select * into v_existing from public.mwd_operations o
    where o.actor_id = v_actor and o.request_id = p_request_id;
  if found then
    if v_existing.kind <> 'contribution'
      or v_existing.source_user_id <> v_actor
      or v_existing.amount_cents <> p_amount_cents then
      raise exception 'Identificador ya usado para otra operacion';
    end if;
    return jsonb_build_object('status', 'already_processed',
      'operation_id', v_existing.id, 'balance_cents', v_balance);
  end if;

  if v_balance < p_amount_cents then
    raise exception 'Saldo ficticio insuficiente';
  end if;

  update public.mwd_wallets
    set balance_cents = balance_cents - p_amount_cents
    where user_id = v_actor;

  insert into public.mwd_operations
    (actor_id, request_id, kind, source_user_id, amount_cents)
    values (v_actor, p_request_id, 'contribution', v_actor, p_amount_cents)
    returning id into v_operation_id;

  return jsonb_build_object('status', 'ok',
    'operation_id', v_operation_id, 'balance_cents', v_balance - p_amount_cents);
end
$$;

returns jsonb language plpgsql security definer set search_path = '' as $$
declare v_request public.mwd_payment_requests%rowtype; v_name text;
begin
 if auth.uid() is null then raise exception 'Debes iniciar sesion'; end if;
 select * into v_request from public.mwd_payment_requests where id=p_payment_id;
 if not found then raise exception 'Solicitud no disponible'; end if;

 return jsonb_build_object('id',v_request.id,'business_name',v_name,'amount_cents',v_request.amount_cents,'currency_code',v_request.currency_code,'status',v_request.status,'description',v_request.description);

create or replace function public.mwd_pay_merchant_demo(p_payment_id uuid)
returns jsonb language plpgsql security definer set search_path = '' as $$
declare v_actor uuid := auth.uid(); v_request public.mwd_payment_requests%rowtype; v_code text; v_result jsonb;
begin
 if v_actor is null then raise exception 'Debes iniciar sesion'; end if;
 if p_payment_id is null then raise exception 'Solicitud invalida'; end if;
 select * into v_request from public.mwd_payment_requests where id=p_payment_id for update;
 if not found then raise exception 'Solicitud no disponible'; end if;
 if v_request.merchant_user_id=v_actor then raise exception 'No puedes pagarte a ti mismo'; end if;
 if v_request.currency_code <> 'MWD' then raise exception 'Solo pagos de prueba'; end if;
 if v_request.status = 'paid' then
   if exists(select 1 from public.mwd_operations where actor_id=v_actor and request_id=p_payment_id and destination_user_id=v_request.merchant_user_id and amount_cents=v_request.amount_cents) then
     return jsonb_build_object('status','already_processed','payment_id',p_payment_id);
   end if;
   raise exception 'Solicitud ya pagada';
 end if;
 if v_request.status <> 'pending' then raise exception 'Solicitud no disponible'; end if;
 select account_code into v_code from public.mwd_wallets where user_id=v_request.merchant_user_id;
 if v_code is null then raise exception 'Comercio sin billetera de prueba'; end if;
 v_result := public.mwd_transfer_demo(v_code,v_request.amount_cents,p_payment_id);
 update public.mwd_payment_requests set status='paid' where id=p_payment_id;
 return jsonb_build_object('status','ok','payment_id',p_payment_id,'transfer',v_result);
end $$;
revoke all on function public.mwd_get_payment_request_demo(uuid) from public, anon;
revoke all on function public.mwd_pay_merchant_demo(uuid) from public, anon;

-- ===== 20261012201539 mwd_ledger_block_direct_posting_and_cross_currency =====
CREATE FUNCTION mwd_finance_private.mwd_ledger_insert_must_be_draft()
RETURNS trigger LANGUAGE plpgsql SET search_path = '' AS $$
BEGIN
  IF NEW.status <> 'draft' THEN
    RAISE EXCEPTION 'Una transaccion nueva debe comenzar en estado draft'
      USING ERRCODE='23514';
  END IF;
  RETURN NEW;
END;
$$;

CREATE TRIGGER mwd_ledger_insert_must_be_draft
BEFORE INSERT ON mwd_finance_private.ledger_transactions
FOR EACH ROW EXECUTE FUNCTION mwd_finance_private.mwd_ledger_insert_must_be_draft();

CREATE FUNCTION mwd_finance_private.mwd_ledger_entry_currency_guard()
RETURNS trigger LANGUAGE plpgsql SET search_path = '' AS $$
DECLARE v_account_currency text; v_transaction_currency text;
BEGIN
  SELECT currency_code INTO v_account_currency
  FROM mwd_finance_private.ledger_accounts WHERE id = NEW.account_id;

  SELECT currency_code INTO v_transaction_currency
  FROM mwd_finance_private.ledger_transactions WHERE id = NEW.transaction_id;

  IF v_account_currency IS NOT NULL AND v_transaction_currency IS NOT NULL
    AND v_account_currency <> v_transaction_currency THEN
    RAISE EXCEPTION 'La moneda de la cuenta y la transaccion deben coincidir'
      USING ERRCODE='23514';
  END IF;
  RETURN NEW;
END;
$$;

CREATE TRIGGER mwd_ledger_entry_currency_guard
BEFORE INSERT OR UPDATE OF account_id, transaction_id
ON mwd_finance_private.ledger_entries
FOR EACH ROW EXECUTE FUNCTION mwd_finance_private.mwd_ledger_entry_currency_guard();

CREATE FUNCTION mwd_finance_private.mwd_ledger_currency_immutable_once_used()
RETURNS trigger LANGUAGE plpgsql SET search_path = '' AS $$
BEGIN
  IF NEW.currency_code IS DISTINCT FROM OLD.currency_code THEN
    IF TG_TABLE_NAME = 'ledger_transactions' AND EXISTS (
      SELECT 1 FROM mwd_finance_private.ledger_entries
      WHERE transaction_id = OLD.id
    ) THEN
      RAISE EXCEPTION 'No se puede cambiar la moneda de una transaccion con asientos'
        USING ERRCODE='23514';
    ELSIF TG_TABLE_NAME = 'ledger_accounts' AND EXISTS (
      SELECT 1 FROM mwd_finance_private.ledger_entries
      WHERE account_id = OLD.id
    ) THEN
      RAISE EXCEPTION 'No se puede cambiar la moneda de una cuenta con asientos'
        USING ERRCODE='23514';
    END IF;
  END IF;
  RETURN NEW;
END;
$$;

CREATE TRIGGER mwd_ledger_transaction_currency_guard
BEFORE UPDATE OF currency_code ON mwd_finance_private.ledger_transactions
FOR EACH ROW EXECUTE FUNCTION mwd_finance_private.mwd_ledger_currency_immutable_once_used();

CREATE TRIGGER mwd_ledger_account_currency_guard
BEFORE UPDATE OF currency_code ON mwd_finance_private.ledger_accounts
FOR EACH ROW EXECUTE FUNCTION mwd_finance_private.mwd_ledger_currency_immutable_once_used();

REVOKE ALL ON FUNCTION mwd_finance_private.mwd_ledger_insert_must_be_draft() FROM PUBLIC;

returns jsonb
language sql
stable
security definer
set search_path = ''
as $$
  select case when auth.uid() is null then
    '[]'::jsonb
  else coalesce(pg_catalog.jsonb_agg(pg_catalog.jsonb_build_object(
    'id', h.id,
    'kind', h.kind,
    'source_user_id', h.source_user_id,
    'destination_user_id', h.destination_user_id,
    'amount_cents', h.amount_cents,

create or replace function public.mwd_quote_international_demo(
  p_destination_country text, p_amount_cents bigint, p_method text
)
returns jsonb
language plpgsql stable security definer
set search_path = ''
as $function$
declare
  v_actor uuid := auth.uid();
  v_route mwd_finance_private.demo_fee_routes%rowtype;
  v_fee bigint;
  v_cost bigint;
  v_is_owner boolean;
  v_quote jsonb;
begin
  if v_actor is null then
    raise exception 'Inicia sesión para consultar la simulación'
      using errcode = '42501';
  end if;
  if p_destination_country is null or
     p_destination_country !~ '^[A-Z]{2}$'
     or p_amount_cents is null or p_amount_cents < 100 or p_amount_cents > 10000000
     or p_method not in ('bank','card') or p_method is null then
    raise exception 'Parámetros de simulación inválidos'
      using errcode = '22023';
  end if;

  select * into v_route from mwd_finance_private.demo_fee_routes
    where origin_country = 'US'
      and destination_country = p_destination_country
      and simulator_enabled is true;
  if not found then
    raise exception 'Ruta de simulación no disponible'
      using errcode = '22023';
  end if;

  v_fee := greatest(v_route.min_fee_cents,
    ceil(p_amount_cents::numeric * v_route.fee_bps::numeric / 10000)::bigint);
  v_cost := v_route.provider_cost_cents +
    case when p_method = 'card' then v_route.card_extra_cost_cents else 0 end;

  v_quote := pg_catalog.jsonb_build_object(
    'status', 'simulation_only',
    'origin', 'US',
    'destination', v_route.destination_country,
    'country', v_route.country_name,
    'sourceCurrency', 'USD',
    'destinationCurrency', v_route.payout_currency,
    'method', p_method,
    'amountCents', p_amount_cents,
    'feeCents', v_fee,
    'totalPayCents', p_amount_cents + v_fee,
    'receiverAmountCents',
      case when v_route.payout_currency = 'USD' then p_amount_cents else null end,
    'fxRate', case when v_route.payout_currency = 'USD' then 1 else null end,
    'realPaymentsEnabled', false
  );

  select exists (
    select 1 from public.mwd_admin_roles
    where user_id = v_actor and role = 'owner'
  ) into v_is_owner;
  if v_is_owner then
    v_quote := v_quote || pg_catalog.jsonb_build_object(
      'costEstimateCents', v_cost,
      'illustrativeMarginCents', v_fee - v_cost
    );
  end if;
  return v_quote;
end
$function$;

  from public, anon, authenticated;

create or replace function public.mwd_conversion_quote_demo(
  p_from_asset text, p_amount_micro bigint, p_client_request_id uuid
)
returns jsonb language plpgsql security definer set search_path='' as $fn$
declare
  v_user uuid := auth.uid();
  v_to text;
  v_rate mwd_swap_private.demo_rates%rowtype;
  v_quote mwd_swap_private.demo_quotes%rowtype;
  v_gross bigint;
  v_fee bigint;
  v_net bigint;
begin
  if v_user is null then raise exception 'Inicia sesión' using errcode='42501'; end if;
  if p_from_asset is null or p_from_asset not in ('COIN','STABLE')
     or p_amount_micro is null or p_amount_micro not between 1000000 and 10000000000
     or p_client_request_id is null then
    raise exception 'Solicitud de conversión inválida' using errcode='22023';
  end if;
  v_to := case when p_from_asset='COIN' then 'STABLE' else 'COIN' end;
  perform mwd_swap_private.ensure_demo_balances(v_user);
  select * into v_quote from mwd_swap_private.demo_quotes q
  where q.user_id=v_user and q.client_request_id=p_client_request_id;
  if found then
    if v_quote.from_asset<>p_from_asset or v_quote.amount_in_micro<>p_amount_micro then
      raise exception 'Identificador reutilizado con otros parámetros' using errcode='22023';
    end if;
  else
    select * into v_rate from mwd_swap_private.demo_rates
      where id='sandbox_v1' and active=true;
    if not found then
      raise exception 'Simulador de cotizaciones no disponible' using errcode='22023';
    end if;
    v_gross := case when p_from_asset='COIN'
      then floor(p_amount_micro::numeric * v_rate.stable_per_coin)::bigint
      else floor(p_amount_micro::numeric / v_rate.stable_per_coin)::bigint end;
    v_fee := ceil(v_gross::numeric * v_rate.fee_bps / 10000)::bigint;
    v_net := v_gross-v_fee;
    if v_net<=0 then raise exception 'Importe menor que el mínimo del simulador' using errcode='22023'; end if;
    insert into mwd_swap_private.demo_quotes
      (user_id,client_request_id,from_asset,to_asset,amount_in_micro,
       gross_out_micro,fee_out_micro,amount_out_micro,stable_per_coin,expires_at)
    values (v_user,p_client_request_id,p_from_asset,v_to,p_amount_micro,
       v_gross,v_fee,v_net,v_rate.stable_per_coin,clock_timestamp()+interval '90 seconds')
    on conflict (user_id,client_request_id) do nothing;
    -- Captura simultaneidad de la misma solicitud.
    select * into v_quote from mwd_swap_private.demo_quotes q
    where q.user_id=v_user and q.client_request_id=p_client_request_id;
    if v_quote.from_asset<>p_from_asset or v_quote.amount_in_micro<>p_amount_micro then
      raise exception 'Identificador reutilizado con otros parámetros' using errcode='22023';
    end if;
  end if;
  return jsonb_build_object(
    'quoteId',v_quote.id,'status',v_quote.status,'provider','synthetic_sandbox',
    'fromAsset',v_quote.from_asset,'toAsset',v_quote.to_asset,
    'amountInMicro',v_quote.amount_in_micro,'grossOutMicro',v_quote.gross_out_micro,
    'feeOutMicro',v_quote.fee_out_micro,'amountOutMicro',v_quote.amount_out_micro,
    'stablePerCoin',v_quote.stable_per_coin,'expiresAt',v_quote.expires_at,
    'realPaymentsEnabled',false,'priceIsReal',false
  );
end; $fn$;

create or replace function public.mwd_conversion_confirm_demo(p_quote_id uuid)
returns jsonb language plpgsql security definer set search_path='' as $fn$
declare
  v_user uuid := auth.uid();
  v_quote mwd_swap_private.demo_quotes%rowtype;
  v_conversion_id uuid;
  v_locked integer;
  v_user_from bigint;
  v_user_to bigint;
  v_pool_to bigint;
begin
  if v_user is null then raise exception 'Inicia sesión' using errcode='42501'; end if;
  if p_quote_id is null then raise exception 'Falta cotización' using errcode='22023'; end if;
  select * into v_quote from mwd_swap_private.demo_quotes q
    where q.id=p_quote_id and q.user_id=v_user for update;
  if not found then raise exception 'Cotización no disponible' using errcode='22023'; end if;
  if v_quote.status='executed' then
    select id into v_conversion_id from mwd_swap_private.demo_conversions
      where quote_id=p_quote_id and user_id=v_user;
    if v_conversion_id is null then
      raise exception 'Asiento pendiente de reconciliación' using errcode='55000';
    end if;
    return jsonb_build_object('status','already_processed','conversionId',v_conversion_id,
      'quoteId',p_quote_id,'realPaymentsEnabled',false);
  end if;
  if v_quote.expires_at<=clock_timestamp() then
    raise exception 'Cotización vencida. Solicita otra.' using errcode='22023';
  end if;
  perform mwd_swap_private.ensure_demo_balances(v_user);
  -- Orden global de bloqueos: pool COIN, STABLE, luego usuario COIN, STABLE.
  perform 1 from mwd_swap_private.demo_pool order by asset_code for update;
  get diagnostics v_locked=row_count;
  if v_locked<>2 then raise exception 'Falta saldo de la bolsa de prueba'; end if;
  perform 1 from mwd_swap_private.demo_balances where user_id=v_user
    order by asset_code for update;
  get diagnostics v_locked=row_count;
  if v_locked<>2 then raise exception 'Faltan saldos de demostración'; end if;
  select balance_micro into v_user_from from mwd_swap_private.demo_balances
    where user_id=v_user and asset_code=v_quote.from_asset;
  select balance_micro into v_user_to from mwd_swap_private.demo_balances
    where user_id=v_user and asset_code=v_quote.to_asset;
  select balance_micro into v_pool_to from mwd_swap_private.demo_pool
    where asset_code=v_quote.to_asset;
  if v_user_from<v_quote.amount_in_micro then
    raise exception 'Saldo de prueba insuficiente' using errcode='22023';
  end if;
  if v_pool_to<v_quote.amount_out_micro then
    raise exception 'Liquidez ficticia de prueba insuficiente' using errcode='22023';
  end if;
  if v_user_to>100000000000000-v_quote.amount_out_micro then
    raise exception 'Límite de saldo de prueba' using errcode='22023';
  end if;

  update mwd_swap_private.demo_balances
    set balance_micro=balance_micro-v_quote.amount_in_micro,updated_at=now()
    where user_id=v_user and asset_code=v_quote.from_asset;
  update mwd_swap_private.demo_pool
    set balance_micro=balance_micro+v_quote.amount_in_micro
    where asset_code=v_quote.from_asset;
  update mwd_swap_private.demo_pool
    set balance_micro=balance_micro-v_quote.amount_out_micro
    where asset_code=v_quote.to_asset;
  update mwd_swap_private.demo_balances
    set balance_micro=balance_micro+v_quote.amount_out_micro,updated_at=now()
    where user_id=v_user and asset_code=v_quote.to_asset;

  insert into mwd_swap_private.demo_conversions
    (quote_id,user_id,from_asset,to_asset,amount_in_micro,
     amount_out_micro,fee_out_micro,stable_per_coin)
  values (v_quote.id,v_user,v_quote.from_asset,v_quote.to_asset,v_quote.amount_in_micro,
          v_quote.amount_out_micro,v_quote.fee_out_micro,v_quote.stable_per_coin)
  returning id into v_conversion_id;

  insert into mwd_swap_private.demo_journal
    (conversion_id,asset_code,account_type,delta_micro)
  values (v_conversion_id,v_quote.from_asset,'user',-v_quote.amount_in_micro),
         (v_conversion_id,v_quote.from_asset,'sandbox_pool',v_quote.amount_in_micro),
         (v_conversion_id,v_quote.to_asset,'sandbox_pool',-v_quote.amount_out_micro),
         (v_conversion_id,v_quote.to_asset,'user',v_quote.amount_out_micro);

  update mwd_swap_private.demo_quotes set status='executed',executed_at=now()
    where id=v_quote.id;
  return jsonb_build_object('status','ok','conversionId',v_conversion_id,
    'quoteId',v_quote.id,'fromAsset',v_quote.from_asset,'toAsset',v_quote.to_asset,
    'amountInMicro',v_quote.amount_in_micro,'amountOutMicro',v_quote.amount_out_micro,
    'feeOutMicro',v_quote.fee_out_micro,'realPaymentsEnabled',false);
end; $fn$;

-- Monitor administrativo: cifras agregadas, sin identidad de clientes.
create or replace function public.mwd_owner_conversion_dashboard_demo()
returns jsonb language plpgsql stable security definer set search_path='' as $fn$
declare
  v_actor uuid;
  v_rate mwd_swap_private.demo_rates%rowtype;
  v_quotes bigint;
  v_live_quotes bigint;
  v_conversions bigint;
  v_wallet_count bigint;
  v_pool jsonb;
  v_wallets jsonb;
  v_fees jsonb;
  v_unbalanced bigint;
  v_incomplete bigint;
  v_audit jsonb;
begin
  v_actor := mwd_partners_private.assert_owner();

  select * into v_rate from mwd_swap_private.demo_rates where id='sandbox_v1';
  if not found then
    raise exception 'Política del simulador no configurada' using errcode='55000';
  end if;

  select count(*),count(*) filter(where status='quoted' and expires_at>now())
  into v_quotes,v_live_quotes from mwd_swap_private.demo_quotes;
  select count(*) into v_conversions from mwd_swap_private.demo_conversions;
  select count(distinct user_id) into v_wallet_count from mwd_swap_private.demo_balances;

  select coalesce(jsonb_object_agg(p.asset_code,p.balance_micro),'{}'::jsonb)
    into v_pool from mwd_swap_private.demo_pool p;
  select coalesce(jsonb_object_agg(b.asset_code,b.total_micro),'{}'::jsonb)
    into v_wallets
  from (select asset_code,sum(balance_micro) total_micro
        from mwd_swap_private.demo_balances group by asset_code) b;
  select coalesce(jsonb_object_agg(f.to_asset,f.total_micro),'{}'::jsonb)
    into v_fees
  from (select to_asset,sum(fee_out_micro) total_micro
        from mwd_swap_private.demo_conversions group by to_asset) f;

  select count(*) into v_unbalanced from (
    select conversion_id, asset_code
    from mwd_swap_private.demo_journal
    group by conversion_id,asset_code
    having sum(delta_micro)<>0 or count(*)<>2
  ) bad;
  select count(*) into v_incomplete from mwd_swap_private.demo_conversions c
  where (select count(*) from mwd_swap_private.demo_journal j

-- Cambio supervisado de tarifa ficticia o pausa/reanudación.

create or replace function public.mwd_owner_conversion_policy_demo(
  p_active boolean,
  p_stable_per_coin numeric,
  p_fee_bps integer,
  p_reason text,
  p_expected_updated_at timestamptz

-- Sólo se devuelve el código aleatorio en la primera creación exitosa.
-- Al repetir la solicitud, nunca se vuelve a divulgar el código secreto.
create or replace function public.mwd_pickup_create_demo(
  p_destination_country text,p_amount_cents bigint,p_request_id uuid
) returns jsonb language plpgsql security definer set search_path='' as $fn$
declare
  v_user uuid := auth.uid();
  v_existing mwd_pickup_private.demo_orders%rowtype;
  v_new mwd_pickup_private.demo_orders%rowtype;
  v_code text;
begin
  if v_user is null then
    raise exception 'Inicia sesión' using errcode='42501';
  end if;
  if p_destination_country is null or p_destination_country !~ '^[A-Z]{2}$'
     or p_amount_cents is null or p_amount_cents not between 100 and 100000
     or p_request_id is null then
    raise exception 'Importe o país inválido para simulación' using errcode='22023';
  end if;

  select * into v_existing from mwd_pickup_private.demo_orders o
   where o.sender_user_id=v_user and o.client_request_id=p_request_id;
  if found then
    if v_existing.amount_cents<>p_amount_cents or
       v_existing.destination_country<>p_destination_country then
      raise exception 'Identificador reutilizado con datos diferentes'
        using errcode='22023';
    end if;
    return jsonb_build_object('status','already_created','orderId',v_existing.id,
      'pickupCode',null,'expiresAt',v_existing.expires_at,
      'realPaymentsEnabled',false);
  end if;

  v_code := 'SIM-' || upper(encode(extensions.gen_random_bytes(16),'hex'));
  insert into mwd_pickup_private.demo_orders
    (sender_user_id,client_request_id,destination_country,amount_cents,
     pickup_code_sha256,expires_at)
  values (v_user,p_request_id,p_destination_country,p_amount_cents,
          encode(extensions.digest(v_code,'sha256'),'hex'),
          clock_timestamp()+interval '24 hours')
  on conflict (sender_user_id,client_request_id) do nothing
  returning * into v_new;

  if v_new.id is null then
    select * into v_existing from mwd_pickup_private.demo_orders o
      where o.sender_user_id=v_user and o.client_request_id=p_request_id;
    if v_existing.id is null or v_existing.amount_cents<>p_amount_cents
       or v_existing.destination_country<>p_destination_country then
      raise exception 'Solicitud duplicada inconsistente' using errcode='22023';
    end if;
    return jsonb_build_object('status','already_created','orderId',v_existing.id,
      'pickupCode',null,'expiresAt',v_existing.expires_at,
      'realPaymentsEnabled',false);
  end if;

  insert into mwd_pickup_private.demo_order_events(order_id,event_kind,actor_id)
  values (v_new.id,'created',v_user);
  return jsonb_build_object('status','created','orderId',v_new.id,
    'pickupCode',v_code,'expiresAt',v_new.expires_at,
    'amountCents',p_amount_cents,'destinationCountry',p_destination_country,
    'realPaymentsEnabled',false,'isFunded',false,'canPayOut',false);
end; $fn$;

create or replace function mwd_pickup_private.demo_order_quota_guard()
returns trigger language plpgsql security invoker set search_path='' as $fn$
begin
 if auth.uid() is null or new.sender_user_id is distinct from auth.uid() then
   raise exception 'No autorizado para crear esta solicitud' using errcode='42501';
 end if;
 perform pg_catalog.pg_advisory_xact_lock(891022,
   pg_catalog.hashtext(new.sender_user_id::text));
 if (select count(*) from mwd_pickup_private.demo_orders o
     where o.sender_user_id=new.sender_user_id
     and o.created_at>clock_timestamp()-interval '1 hour') >= 5 then
   raise exception 'Límite horario de solicitudes de prueba' using errcode='22023';
 end if;

-- Moneda Mundial 2.0 · Controles antifraude del retiro SIMULADO.
-- Sin pagos, sin KYC real y sin otorgar poderes a agentes candidatos.
-- Auditoría de intentos SIN guardar códigos ni hashes de códigos.
create table if not exists mwd_pickup_private.demo_claim_attempts (
  id bigint generated always as identity primary key,
  order_id uuid not null references mwd_pickup_private.demo_orders(id),
  actor_id uuid not null references auth.users(id),

-- Ajuste de la función antigua: los intentos incorrectos ahora se AUDITAN.
-- No hacer RAISE después de almacenar un fallo: Postgres lo revertiría.
-- La fila de la orden se bloquea para contabilizar intentos concurrentes.
create or replace function public.mwd_owner_pickup_simulate_demo(
 p_order_id uuid, p_pickup_code text
) returns jsonb language plpgsql security definer set search_path='' as $fn$
declare
  v_actor uuid;
  v_order mwd_pickup_private.demo_orders%rowtype;
  v_clean text := upper(btrim(coalesce(p_pickup_code,'')));
  v_failures integer;
begin
  v_actor := mwd_partners_private.assert_owner();
  if p_order_id is null then
    raise exception 'Solicitud de prueba inválida' using errcode='22023';
  end if;

  select * into v_order from mwd_pickup_private.demo_orders
  where id=p_order_id for update;
  if not found then
    raise exception 'Solicitud de prueba no disponible' using errcode='22023';
  end if;
  -- La orden no puede canjearse dos veces, incluso ante solicitudes simultáneas.
  if v_order.status <> 'pending' or v_order.expires_at <= clock_timestamp() then
    return jsonb_build_object('status','not_available','orderId',v_order.id,
       'cashDisbursed',false,'realPaymentsEnabled',false);
  end if;

  select count(*) into v_failures
  from mwd_pickup_private.demo_claim_attempts a
  where a.order_id=v_order.id and a.outcome='invalid'
    and a.created_at > clock_timestamp()-interval '15 minutes';
  if v_failures >= 5 then
    return jsonb_build_object('status','temporarily_blocked','orderId',v_order.id,
      'message','Demasiados códigos incorrectos. Intenta más tarde.',
      'cashDisbursed',false,'realPaymentsEnabled',false);
  end if;

  -- El código no se registra, ni se muestra en auditoría.
  if length(v_clean)>80 or v_clean !~ '^SIM-[A-F0-9]{32}$'
     or v_order.pickup_code_sha256 <>
        encode(extensions.digest(v_clean,'sha256'),'hex') then
    insert into mwd_pickup_private.demo_claim_attempts
       (order_id,actor_id,outcome)
    values(v_order.id,v_actor,'invalid');
    return jsonb_build_object('status','invalid_code',
      'orderId',v_order.id,'remainingAttempts',greatest(0,4-v_failures),
      'cashDisbursed',false,'realPaymentsEnabled',false);
  end if;

  update mwd_pickup_private.demo_orders
   set status='simulated_collected',
       processed_at=clock_timestamp(),processed_by=v_actor
   where id=v_order.id and status='pending';
  insert into mwd_pickup_private.demo_order_events(order_id,event_kind,actor_id)
  values(v_order.id,'simulated_collected',v_actor);
  insert into mwd_pickup_private.demo_claim_attempts(order_id,actor_id,outcome)
  values(v_order.id,v_actor,'successful');
  return jsonb_build_object('status','simulated_collected','orderId',v_order.id,
    'cashDisbursed',false,'realPaymentsEnabled',false);
end; $fn$;

-- Reporte limitado para el propietario. Jamás incluye el código ingresado.
create or replace function public.mwd_owner_pickup_security_demo()
returns jsonb language plpgsql stable security definer set search_path='' as $fn$
declare
  v_recent jsonb;
  v_locked integer;
  v_failed_hour integer;
begin
  perform mwd_partners_private.assert_owner();

  select count(*) into v_failed_hour
  from mwd_pickup_private.demo_claim_attempts
  where outcome='invalid' and created_at>clock_timestamp()-interval '1 hour';

  select count(*) into v_locked from (
    select order_id from mwd_pickup_private.demo_claim_attempts
    where outcome='invalid' and created_at>clock_timestamp()-interval '15 minutes'
    group by order_id having count(*)>=5
  ) x;

  select coalesce(jsonb_agg(jsonb_build_object(
    'orderId',a.order_id,'outcome',a.outcome,'time',a.created_at,
    'country',o.destination_country)
    order by a.created_at desc,a.id desc),'[]'::jsonb)
  into v_recent
  from (select * from mwd_pickup_private.demo_claim_attempts
        order by created_at desc,id desc limit 30) a
  join mwd_pickup_private.demo_orders o on o.id=a.order_id;

  return jsonb_build_object(
    'mode','demo_only','realPaymentsEnabled',false,
    'invalidAttemptsLastHour',v_failed_hour,
    'temporarilyBlockedOrders',v_locked,
    'recentAttempts',v_recent,
    'secretCodesLogged',false);
end; $fn$;

create or replace function public.mwd_owner_partner_due_diligence_save_demo(
  p_partner_id uuid,
  p_requirement_type text,
  p_review_status text,
  p_valid_until date default null
) returns jsonb language plpgsql security definer set search_path = '' as $$
declare v_actor uuid;
begin
  v_actor := mwd_partners_private.assert_owner();
  if p_partner_id is null or p_requirement_type is null
     or p_requirement_type not in (
       'legal_registration','regulatory_license','jurisdiction_permissions',
       'signed_agreement','aml_kyc_program','sanctions_controls',
       'consumer_protection','data_security','settlement_controls','tax_reporting'
     )
     or p_review_status is null or p_review_status not in (
       'not_started','requested','received_unverified','needs_changes','reviewed_internal'
     )
     or (p_review_status = 'not_started' and p_valid_until is not null)
     or (p_valid_until is not null and p_valid_until < date '1900-01-01') then
    raise exception 'Datos de revision invalidos' using errcode='22023';
  end if;

  -- Bloqueo en el mismo orden para impedir nuevas evaluaciones al archivarse.
  perform 1 from mwd_partners_private.partners p
   where p.id = p_partner_id and p.review_status <> 'archived'
   for update;
  if not found then
    raise exception 'Socio inexistente o archivado' using errcode='22023';
  end if;

  insert into mwd_partners_private.due_diligence_reviews
    (partner_id,requirement_type,review_status,valid_until,created_by,updated_by)
  values (p_partner_id,p_requirement_type,p_review_status,p_valid_until,v_actor,v_actor)
  on conflict on constraint mwd_partner_due_diligence_unique
    do update set
      review_status = excluded.review_status,
      valid_until = excluded.valid_until,
      updated_by = v_actor,
      updated_at = now();

  return jsonb_build_object(
    'saved',true,
    'requirement_type',p_requirement_type,
    'review_status',p_review_status,
    'real_payouts_enabled',false,
    'real_partners_approved',false
  );
end; $$;

-- ===== 20261016194207 mwd_owner_partner_expiry_dashboard_demo_20261016 =====
-- Reporte de vencimientos de candidatos: SOLO METADATOS, SIN SERVICIO FINANCIERO.
-- Requiere mwd_partners_private.{partners,due_diligence_reviews}
-- No certifica licencias ni genera pagos.
create or replace function public.mwd_owner_partner_expiry_dashboard_demo(
 p_days integer default 60
) returns table(
 partner_id uuid, partner_name text, country_code text, partner_status text,
 requirement_type text, review_status text, valid_until date,
 days_remaining integer, alert_level text
) language plpgsql security definer set search_path='' as $$
begin
 perform mwd_partners_private.assert_owner();
 if p_days is null or p_days not between 30 and 365 then
   raise exception 'Ventana inválida: utilizar entre 30 y 365 días' using errcode='22023';
 end if;
 return query
 select p.id,p.legal_name,p.country_code,p.review_status,
        req.type,
        coalesce(r.review_status,'not_started'::text),
        r.valid_until,
        case when r.valid_until is null then null::integer
             else (r.valid_until - current_date)::integer end,
        case when r.valid_until is null then 'missing_date'::text
             when r.valid_until < current_date then 'expired'::text
             when r.valid_until <= (current_date + 30) then 'due_soon'::text
             when r.valid_until <= (current_date + p_days) then 'due_within_window'::text
             else 'later'::text end
 from mwd_partners_private.partners p
 cross join (values
   ('regulatory_license'::text),('signed_agreement'::text),
   ('jurisdiction_permissions'::text)
 ) req(type)
 left join mwd_partners_private.due_diligence_reviews r
   on r.partner_id=p.id and r.requirement_type=req.type
 where p.review_status<>'archived'
 order by
   case when r.valid_until < current_date then 0
        when r.valid_until <= (current_date + 30) then 1
        when r.valid_until <= (current_date + p_days) then 2
        when r.valid_until is null then 3 else 4 end,
   r.valid_until nulls last,p.legal_name,p.id,req.type
 limit 600;
end; $$;

-- ===== 20261016205447 mwd_partner_readiness_overview_internal_demo =====
-- Tablero exclusivamente interno de CANDIDATOS. NO permite habilitar dinero real.
-- SQL para Supabase y PostgreSQL aislado; no crea, cambia ni aprueba socios.
create or replace function public.mwd_owner_partner_readiness_overview_demo(
  p_country_code text default null
) returns table(
  partner_id uuid,
  partner_name text,
  country_code text,
  partner_kind text,
  partner_status text,
  branch_drafts integer,
  requirement_count integer,
  pending_requirements integer,
  internally_reviewed integer,
  expired_requirements integer,
  due_30_requirements integer,
  missing_critical_dates integer,
  real_payments_enabled boolean
) language plpgsql security definer set search_path = '' as $$
begin
  perform mwd_partners_private.assert_owner();
  if p_country_code is not null and p_country_code !~ '^[A-Z]{2}$' then
    raise exception 'Código de país inválido' using errcode='22023';
  end if;
  return query
    select p.id, p.legal_name, p.country_code, p.partner_kind, p.review_status,
           coalesce(branches.count_branches, 0)::integer,
           coalesce(progress.all_count, 0)::integer,
           coalesce(progress.pending_count, 0)::integer,
           coalesce(progress.reviewed_count, 0)::integer,
           coalesce(progress.expired_count, 0)::integer,
           coalesce(progress.due_soon_count, 0)::integer,
           coalesce(progress.missing_dates_count, 0)::integer,
           false
    from mwd_partners_private.partners p
    left join lateral (
      select count(*)::integer count_branches
      from mwd_partners_private.locations l where l.partner_id = p.id
    ) branches on true
    left join lateral (
      select
        count(*)::integer all_count,
        count(*) filter (
          where coalesce(d.review_status, 'not_started') <> 'reviewed_internal'
             or (d.valid_until is not null and d.valid_until < current_date)
        )::integer pending_count,
        count(*) filter (
          where d.review_status = 'reviewed_internal'
            and (d.valid_until is null or d.valid_until >= current_date)
        )::integer reviewed_count,
        count(*) filter (
          where d.valid_until < current_date
        )::integer expired_count,
        count(*) filter (
          where d.valid_until between current_date and current_date+30
        )::integer due_soon_count,
        count(*) filter (
          where types.kind in ('regulatory_license','jurisdiction_permissions','signed_agreement')
            and d.valid_until is null
        )::integer missing_dates_count
      from (values
        ('legal_registration'::text), ('regulatory_license'), ('jurisdiction_permissions'),
        ('signed_agreement'), ('aml_kyc_program'), ('sanctions_controls'),
        ('consumer_protection'), ('data_security'), ('settlement_controls'), ('tax_reporting')
      ) types(kind)
      left join mwd_partners_private.due_diligence_reviews d
        on d.partner_id = p.id and d.requirement_type = types.kind
    ) progress on true
    where p.review_status <> 'archived'
      and (p_country_code is null or p.country_code = p_country_code)
    order by p.country_code, p.legal_name, p.id
    limit 1000;
end; $$;

create or replace function mwd_remittance_demo_private.assert_owner()
returns uuid language plpgsql security definer set search_path = '' as $$
declare v_actor uuid := auth.uid();
begin
  if v_actor is null or not exists (
    select 1 from public.mwd_admin_roles r
    where r.user_id=v_actor and r.role='owner'
  ) then
    raise exception 'Acceso exclusivo del propietario a remesas ficticias'
      using errcode='42501';
  end if;
  return v_actor;
end; $$;

create or replace function public.mwd_owner_remittance_create_demo(
  p_destination_country text, p_amount_cents bigint, p_request_id uuid
) returns jsonb language plpgsql security definer set search_path = '' as $$
declare
 v_actor uuid;
 v_id uuid;
 v_existing mwd_remittance_demo_private.orders%rowtype;
begin
 v_actor := mwd_remittance_demo_private.assert_owner();
 if p_request_id is null or p_destination_country is null or
    p_destination_country !~ '^[A-Z]{2}$' or
    p_amount_cents is null or p_amount_cents not between 100 and 10000000 then
   raise exception 'Datos de simulación de remesa no válidos' using errcode='22023';
 end if;
 insert into mwd_remittance_demo_private.orders
   (created_by,request_id,destination_country,amount_cents)
 values (v_actor,p_request_id,p_destination_country,p_amount_cents)
 on conflict (created_by,request_id) do nothing returning id into v_id;

 if v_id is null then
   select * into v_existing
     from mwd_remittance_demo_private.orders
     where created_by=v_actor and request_id=p_request_id;
   if not found then raise exception 'No se pudo resolver el reintento' using errcode='40001'; end if;
   if v_existing.destination_country is distinct from p_destination_country
      or v_existing.amount_cents is distinct from p_amount_cents then
     raise exception 'Clave de reintento utilizada con otros datos'
       using errcode='22023';
   end if;
   return jsonb_build_object('id',v_existing.id,'status',v_existing.status,
     'already_processed',true,'real_payments_enabled',false,'redeemable',false);
 end if;

 insert into mwd_remittance_demo_private.events
   (order_id,actor_id,request_id,event_type,from_status,to_status)
 values (v_id,v_actor,p_request_id,'created',null,'draft');
 return jsonb_build_object('id',v_id,'status','draft',
    'already_processed',false,'real_payments_enabled',false,'redeemable',false);
end; $$;

create or replace function public.mwd_owner_remittance_transition_demo(
  p_order_id uuid,p_new_status text,p_request_id uuid
) returns jsonb language plpgsql security definer set search_path = '' as $$
declare
 v_actor uuid;
 v_order mwd_remittance_demo_private.orders%rowtype;
 v_previous mwd_remittance_demo_private.events%rowtype;
 v_event_type text;
begin
 v_actor := mwd_remittance_demo_private.assert_owner();
 if p_order_id is null or p_request_id is null or
    p_new_status is null or p_new_status not in ('review','simulated','cancelled')
 then raise exception 'Transición ficticia no válida' using errcode='22023'; end if;

 select * into v_order from mwd_remittance_demo_private.orders
   where id=p_order_id for update;
 if not found then raise exception 'Remesa ficticia inexistente' using errcode='22023'; end if;

 select * into v_previous from mwd_remittance_demo_private.events
   where order_id=p_order_id and request_id=p_request_id;
 if found then
   if v_previous.to_status <> p_new_status then
     raise exception 'Clave de reintento utilizada con otro estado'
       using errcode='22023';
   end if;
   return jsonb_build_object('id',p_order_id,'status',v_order.status,
     'already_processed',true,'real_payments_enabled',false,'redeemable',false);
 end if;

 if not (
   (v_order.status='draft' and p_new_status in ('review','cancelled')) or
   (v_order.status='review' and p_new_status in ('simulated','cancelled'))
 ) then
   raise exception 'El cambio de estado no está permitido' using errcode='22023';
 end if;
 v_event_type := case p_new_status
   when 'review' then 'submitted_for_review'
   when 'simulated' then 'simulation_completed'
   else 'cancelled' end;
 update mwd_remittance_demo_private.orders
    set status=p_new_status, updated_at=now()
   where id=p_order_id;
 insert into mwd_remittance_demo_private.events
   (order_id,actor_id,request_id,event_type,from_status,to_status)
 values (p_order_id,v_actor,p_request_id,v_event_type,v_order.status,p_new_status);
 return jsonb_build_object('id',p_order_id,'status',p_new_status,
    'already_processed',false,'real_payments_enabled',false,'redeemable',false);
end; $$;

create or replace function public.mwd_owner_assign_payout_design_demo(
 p_order_id uuid,p_partner_id uuid,p_location_id uuid,p_request_id uuid
) returns jsonb language plpgsql security definer set search_path='' as $$
declare
 v_actor uuid;
 v_order mwd_remittance_demo_private.orders%rowtype;

 v_location mwd_partners_private.locations%rowtype;
 v_existing mwd_remittance_demo_private.payout_assignments%rowtype;

begin
 v_actor:=mwd_remittance_demo_private.assert_owner();
 if p_order_id is null or p_partner_id is null or p_location_id is null
    or p_request_id is null then
    raise exception 'Parámetros de asociación ficticia inválidos' using errcode='22023';
 end if;

 -- Bloqueo de orden: evita dos asignaciones concurrentes al mismo envío.
 select * into v_order from mwd_remittance_demo_private.orders

       'already_processed',true,'real_payments_enabled',false,'redeemable',false);

   raise exception 'La remesa ficticia debe estar en revisión' using errcode='22023';

     using errcode='22023';

   raise exception 'Sucursal propuesta inválida para el candidato'
      using errcode='22023';

   order_id,partner_id,location_id,currency_code,created_by,assignment_request_id)

   assignment_id,actor_id,request_id,event_type)
 values(v_id,v_actor,p_request_id,'assigned');
 return jsonb_build_object('id',v_id,'status','planned',
   'already_processed',false,'real_payments_enabled',false,'redeemable',false);
end; $$;

-- Un intento incorrecto debe devolver resultado, no lanzar excepción,
-- porque un RAISE EXCEPTION revertiría el contador de intentos.
create or replace function public.mwd_owner_try_pickup_code_demo(
 p_order_id uuid,p_demo_code text,p_request_id uuid
) returns jsonb language plpgsql security definer set search_path='' as $$
declare
 v_actor uuid;
 v_sample mwd_remittance_demo_private.pickup_code_samples%rowtype;
 v_cached jsonb;
 v_result jsonb;
 v_hash text;
 v_tries integer;
 v_new_status text;
begin
 v_actor:=mwd_remittance_demo_private.assert_owner();
 if p_order_id is null or p_request_id is null then
   raise exception 'Parámetros inválidos' using errcode='22023'; end if;
 select c.* into v_sample from mwd_remittance_demo_private.orders o
 join mwd_remittance_demo_private.payout_assignments a on a.order_id=o.id
 join mwd_remittance_demo_private.pickup_code_samples c on c.assignment_id=a.id
 where o.id=p_order_id for update of c;
 if not found then raise exception 'No existe código ficticio' using errcode='22023'; end if;
 -- La fila se bloquea FOR UPDATE antes del chequeo, serializando solicitudes
 -- simultáneas. Un reintento exacto no suma un segundo intento fallido.
 select a.result into v_cached
 from mwd_remittance_demo_private.pickup_code_attempts a
 where a.sample_id=v_sample.id and a.request_id=p_request_id;
 if found then return v_cached || jsonb_build_object('already_processed',true); end if;
 if v_sample.status<>'active' then
   return jsonb_build_object('status',v_sample.status,'accepted',false,
     'real_payments_enabled',false,'redeemable',false);
 end if;

   update mwd_remittance_demo_private.pickup_code_samples
   set status='expired',closed_at=now() where id=v_sample.id;
   insert into mwd_remittance_demo_private.pickup_code_events(sample_id,actor_id,action)
   values(v_sample.id,v_actor,'expired');
   return jsonb_build_object('status','expired','accepted',false,
     'real_payments_enabled',false,'redeemable',false);
 end if;

 -- Se compara sólo en servidor, sin exponer hash ni código almacenado.
 if p_demo_code is null or p_demo_code !~ '^DEMO-[0-9A-F]{32}$' then
   v_hash:=null;
 else

   update mwd_remittance_demo_private.pickup_code_samples

   insert into mwd_remittance_demo_private.pickup_code_events(sample_id,actor_id,action)
   values(v_sample.id,v_actor,'wrong_code');
   if v_new_status='locked' then
     insert into mwd_remittance_demo_private.pickup_code_events(sample_id,actor_id,action)
     values(v_sample.id,v_actor,'locked');
   end if;
   v_result:=jsonb_build_object('status',v_new_status,'accepted',false,
     'remaining_attempts',5-v_tries,'real_payments_enabled',false,'redeemable',false);
   insert into mwd_remittance_demo_private.pickup_code_attempts(sample_id,request_id,result)
   values(v_sample.id,p_request_id,v_result);
   return v_result;
 end if;
 if v_sample.identity_check_simulated_at is null then
   v_result:=jsonb_build_object('status','identity_demo_pending','accepted',false,
     'identity_real_verified',false,'real_payments_enabled',false,'redeemable',false);
   insert into mwd_remittance_demo_private.pickup_code_attempts(sample_id,request_id,result)
   values(v_sample.id,p_request_id,v_result);
   return v_result;
 end if;

 update mwd_remittance_demo_private.pickup_code_samples
 set status='consumed',consume_request_id=p_request_id,closed_at=now()
 where id=v_sample.id;
 insert into mwd_remittance_demo_private.pickup_code_events(sample_id,actor_id,action)
 values(v_sample.id,v_actor,'consumed');
 -- La simulación NO cambia el libro contable ni registra pagos de efectivo.
 v_result:=jsonb_build_object('status','consumed','accepted',true,
   'sample_completed',true,'identity_real_verified',false,
   'money_paid',false,'real_payments_enabled',false,'redeemable',false);
 insert into mwd_remittance_demo_private.pickup_code_attempts(sample_id,request_id,result)
 values(v_sample.id,p_request_id,v_result);
 return v_result;
end; $$;

) returns table(
 order_id uuid,source_country text,destination_country text,
 amount_cents bigint,order_status text,created_at timestamptz,
 event_count bigint,assignment_status text,pickup_status text,
 failed_attempts integer,expires_at timestamptz,identity_demo_checked boolean,
 anomaly_count integer,attention text,real_payments_enabled boolean
) language plpgsql security definer set search_path='' as $$
begin

 if (p_country_code is not null and p_country_code !~ '^[A-Z]{2}$')

  raise exception 'Filtro administrativo inválido' using errcode='22023';

 return query
 select o.id,o.source_country,o.destination_country,o.amount_cents,
  o.status,o.created_at,
  (select count(*) from mwd_remittance_demo_private.events ev where ev.order_id=o.id),
  coalesce(a.status,'unassigned'),
  coalesce(case when c.status='active' and now()>=c.expires_at
    then 'expired' else c.status end,'not_issued'),
  coalesce(c.failed_attempts,0),c.expires_at,
  coalesce(c.identity_check_simulated_at is not null,false),
  ((case when c.failed_attempts>=3 then 1 else 0 end)+
   (case when c.status='locked' then 1 else 0 end)+
   (case when o.status='cancelled' and c.status='active' then 1 else 0 end))::integer,
  case when c.status='locked' or c.failed_attempts>=3 or
                 (o.status='cancelled' and c.status='active') then 'review'

 left join mwd_remittance_demo_private.payout_assignments a on a.order_id=o.id
 left join mwd_remittance_demo_private.pickup_code_samples c on c.assignment_id=a.id

-- ===== 20261018224050 mwd_conversion_reconcile_demo_read_only =====
-- Moneda Mundial 2.0 | Consultar SIN REEJECUTAR una conversión ficticia incierta.
-- Implementación propuesta: no ejecutar en producción ni habilitar dinero real.
-- Requiere haber instalado database/mwd_coin_stable_conversion_demo.sql.
-- La fila de cotización se bloquea para serializar el resultado con la confirmación.
create or replace function public.mwd_conversion_reconcile_demo(p_quote_id uuid)
returns jsonb
language plpgsql security definer set search_path = ''
as $fn$
declare
  v_user uuid := auth.uid();
  v_quote mwd_swap_private.demo_quotes%rowtype;
  v_conversion mwd_swap_private.demo_conversions%rowtype;
  v_state text;
begin
  if v_user is null then
    raise exception 'Sesión requerida' using errcode='42501';
  end if;
  if p_quote_id is null then
    raise exception 'Referencia inválida' using errcode='22023';
  end if;

  -- Impide inspeccionar transacciones ajenas y observa los commits de confirmación.
  select * into v_quote
    from mwd_swap_private.demo_quotes
    where id=p_quote_id and user_id=v_user
    for update;
  if not found then
    raise exception 'Referencia no disponible' using errcode='22023';
  end if;

  if v_quote.status='executed' then
    select * into v_conversion
      from mwd_swap_private.demo_conversions
      where quote_id=p_quote_id and user_id=v_user;
    if not found then
      raise exception 'Conciliación pendiente de revisión' using errcode='55000';
    end if;
    v_state:='executed';
  elsif v_quote.status='quoted' and v_quote.expires_at<=clock_timestamp() then
    -- Una cotización no ejecutada y ya vencida no puede ser confirmada después.
    v_state:='expired';
  elsif v_quote.status='quoted' then
    v_state:='quoted';
  else
    raise exception 'Estado desconocido' using errcode='55000';
  end if;

  return jsonb_build_object(
    'mode','sandbox_only','realPaymentsEnabled',false,
    'status',v_state,'quoteId',v_quote.id,
    'fromAsset',v_quote.from_asset,'toAsset',v_quote.to_asset,
    'amountInMicro',v_quote.amount_in_micro,
    'amountOutMicro',v_quote.amount_out_micro,
    'feeOutMicro',v_quote.fee_out_micro,
    'conversionId',case when v_state='executed' then v_conversion.id else null end
  );
end;
$fn$;

-- La función pública tiene acceso por RPC únicamente con token autenticado.
-- El rol autenticado NO obtiene acceso directo al esquema privado.
revoke all on function public.mwd_conversion_reconcile_demo(uuid) from public, anon, authenticated;
grant execute on function public.mwd_conversion_reconcile_demo(uuid) to authenticated;


-- ===== 20261018235651 mwd_operation_status_read_only_demo =====
-- Moneda Mundial 2.0 • Consulta de estado de operaciones MWD ficticias.
-- Nunca crea, reintenta, debita, acredita ni cancela una transferencia.
-- UNCONFIRMED NO significa que la operación haya fallado: podría seguir en tránsito.
create or replace function public.mwd_operation_status_demo(p_request_id uuid)
returns jsonb language plpgsql security definer set search_path=''
as $fn$
declare
  v_actor uuid := auth.uid();
  v_op public.mwd_operations%rowtype;
  v_code text;
begin
  if v_actor is null then
    raise exception 'Se requiere una sesión autenticada' using errcode='42501';
  end if;
  if p_request_id is null then
    raise exception 'Referencia inválida' using errcode='22023';
  end if;
  select * into v_op from public.mwd_operations
   where actor_id=v_actor and request_id=p_request_id;
  if not found then
    return jsonb_build_object('mode','sandbox_only','realPaymentsEnabled',false,
      'status','unconfirmed','requestId',p_request_id);
  end if;
  if v_op.kind not in ('transfer','contribution') or
     v_op.source_user_id is distinct from v_actor then
    raise exception 'Registro requiere revisión' using errcode='55000';
  end if;
  if v_op.kind='transfer' then
    select account_code into v_code from public.mwd_wallets
     where user_id=v_op.destination_user_id;
    if v_code is null then
      raise exception 'Destino no verificable' using errcode='55000';
    end if;
  end if;
  return jsonb_build_object(
    'mode','sandbox_only','realPaymentsEnabled',false,
    'status','processed','requestId',p_request_id,
    'operationId',v_op.id,'kind',v_op.kind,'amountCents',v_op.amount_cents,
    'recipientCode',v_code,'createdAt',v_op.created_at
  );
end;
$fn$;

grant execute on function public.mwd_operation_status_demo(uuid) to authenticated;

CREATE OR REPLACE FUNCTION public.mwd_pickup_create_demo(p_destination_country text, p_amount_cents bigint, p_request_id uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare
  v_user uuid := auth.uid();
  v_existing mwd_pickup_private.demo_orders%rowtype;
  v_new mwd_pickup_private.demo_orders%rowtype;
  v_code text;
begin
  if v_user is null then
    raise exception 'Inicia sesión' using errcode='42501';
  end if;
  if p_destination_country is null or p_destination_country !~ '^[A-Z]{2}$'
     or p_amount_cents is null or p_amount_cents not between 100 and 100000
     or p_request_id is null then
    raise exception 'Importe o país inválido para simulación' using errcode='22023';
  end if;

  -- Serialize new requests per user to prevent parallel limit bypass.
  perform pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(v_user::text, 4107));

  select * into v_existing from mwd_pickup_private.demo_orders o
   where o.sender_user_id=v_user and o.client_request_id=p_request_id;
  if found then
    if v_existing.amount_cents<>p_amount_cents or
       v_existing.destination_country<>p_destination_country then
      raise exception 'Identificador reutilizado con datos diferentes'
        using errcode='22023';
    end if;
    return jsonb_build_object('status','already_created','orderId',v_existing.id,
      'pickupCode',null,'expiresAt',v_existing.expires_at,
      'realPaymentsEnabled',false);
  end if;

  if (select count(*) from mwd_pickup_private.demo_orders o
      where o.sender_user_id=v_user
        and o.created_at > clock_timestamp()-interval '1 hour') >= 5 then
    raise exception 'Límite de 5 órdenes ficticias por hora. Intenta más tarde.'
      using errcode='P0001';
  end if;

  v_code := 'SIM-' || upper(encode(extensions.gen_random_bytes(16),'hex'));
  insert into mwd_pickup_private.demo_orders
    (sender_user_id,client_request_id,destination_country,amount_cents,
     pickup_code_sha256,expires_at)
  values (v_user,p_request_id,p_destination_country,p_amount_cents,
          encode(extensions.digest(v_code,'sha256'),'hex'),
          clock_timestamp()+interval '24 hours')
  on conflict (sender_user_id,client_request_id) do nothing
  returning * into v_new;

  if v_new.id is null then
    select * into v_existing from mwd_pickup_private.demo_orders o
      where o.sender_user_id=v_user and o.client_request_id=p_request_id;
    if v_existing.id is null or v_existing.amount_cents<>p_amount_cents
       or v_existing.destination_country<>p_destination_country then
      raise exception 'Solicitud duplicada inconsistente' using errcode='22023';
    end if;
    return jsonb_build_object('status','already_created','orderId',v_existing.id,
      'pickupCode',null,'expiresAt',v_existing.expires_at,
      'realPaymentsEnabled',false);
  end if;

  insert into mwd_pickup_private.demo_order_events(order_id,event_kind,actor_id)
  values (v_new.id,'created',v_user);
  return jsonb_build_object('status','created','orderId',v_new.id,
    'pickupCode',v_code,'expiresAt',v_new.expires_at,
    'amountCents',p_amount_cents,'destinationCountry',p_destination_country,
    'realPaymentsEnabled',false,'isFunded',false,'canPayOut',false);
end; $function$
;


-- ===== 20261126024008 mwd_demo_contribution_ledger_closure =====
set local lock_timeout='5s';
lock table public.mwd_wallets in share row exclusive mode;
lock table public.mwd_operations in share row exclusive mode;
alter table mwd_finance_private.ledger_transactions drop constraint ledger_transactions_transaction_kind_check;
alter table mwd_finance_private.ledger_transactions add constraint ledger_transactions_transaction_kind_check check (transaction_kind in ('deposit','withdrawal','wallet_transfer','direct_transfer','fee','reversal','demo_contribution'));
create or replace function mwd_finance_private.post_demo_contribution_v1(p_id uuid)
returns void language plpgsql security definer set search_path='' as $$
declare o public.mwd_operations%rowtype; src uuid; dst uuid; tx uuid;
begin
 select * into strict o from public.mwd_operations where id=p_id and kind='contribution';
 if o.source_user_id is null or o.source_user_id is distinct from o.actor_id or o.amount_cents is null or o.amount_cents<=0 then raise exception 'Invalid demo contribution'; end if;
 insert into mwd_finance_private.ledger_accounts(account_code,owner_user_id,currency_code,account_type)
 values ('MWD-DEMO-USER-'||o.source_user_id,o.source_user_id,'MWD','customer_liability'),('MWD-DEMO-CONTRIBUTION-CLEARING',null,'MWD','suspense') on conflict(account_code) do nothing;
 select id into strict src from mwd_finance_private.ledger_accounts where account_code='MWD-DEMO-USER-'||o.source_user_id and owner_user_id=o.source_user_id and currency_code='MWD' and account_type='customer_liability';
 select id into strict dst from mwd_finance_private.ledger_accounts where account_code='MWD-DEMO-CONTRIBUTION-CLEARING' and owner_user_id is null and currency_code='MWD' and account_type='suspense';
 insert into mwd_finance_private.ledger_transactions(external_reference,transaction_kind,currency_code,status) values ('mwd-demo-contribution:'||o.id,'demo_contribution','MWD','draft') on conflict(external_reference) do nothing returning id into tx;
 if tx is null then
 select id into strict tx from mwd_finance_private.ledger_transactions where external_reference='mwd-demo-contribution:'||o.id and transaction_kind='demo_contribution' and currency_code='MWD' and status='posted';
 if (select count(*) from mwd_finance_private.ledger_entries where transaction_id=tx)<>2 or not exists(select 1 from mwd_finance_private.ledger_entries where transaction_id=tx and account_id=src and entry_side='debit' and amount_minor=o.amount_cents) or not exists(select 1 from mwd_finance_private.ledger_entries where transaction_id=tx and account_id=dst and entry_side='credit' and amount_minor=o.amount_cents) then raise exception 'Existing contribution mismatch'; end if;
 return;
 end if;

 update mwd_finance_private.ledger_transactions set status='posted' where id=tx;
end $$;

create or replace view mwd_finance_private.demo_ledger_alerts_v1 with (security_invoker=true) as
select o.id as operation_id, case when t.id is null then 'missing_ledger' when t.status <> 'posted' then 'not_posted' when count(e.id) <> 2 then 'wrong_entry_count' when coalesce(sum(e.amount_minor) filter(where e.entry_side='debit'),0) <> o.amount_cents or coalesce(sum(e.amount_minor) filter(where e.entry_side='credit'),0) <> o.amount_cents then 'amount_mismatch' else 'ok' end as alert_status
from public.mwd_operations o left join mwd_finance_private.ledger_transactions t on t.external_reference=(case when o.kind='transfer' then 'mwd-demo-transfer:' else 'mwd-demo-contribution:' end)||o.id::text left join mwd_finance_private.ledger_entries e on e.transaction_id=t.id where o.kind in ('transfer','contribution') group by o.id,o.amount_cents,t.id,t.status;
revoke all on mwd_finance_private.demo_ledger_alerts_v1 from public,anon,authenticated;
select alert_status,count(*) from mwd_finance_private.demo_ledger_alerts_v1 group by alert_status;
